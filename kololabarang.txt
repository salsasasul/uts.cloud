  <script>
    const SUPABASE_URL = "https://bqffaoqdwqinlonmuzyr.supabase.co";
    const SUPABASE_KEY = "sb_publishable_7GLcIpYrQYhLLHqQG-ytmA_Nljw7tFO";
    const TABLE_ENDPOINT = `${SUPABASE_URL}/rest/v1/barang`;

    const headers = {
      "apikey": SUPABASE_KEY,
      "Authorization": `Bearer ${SUPABASE_KEY}`,
      "Content-Type": "application/json",
      "Prefer": "return=representation"
    };

    let daftarBarang = [];

    function setStatus(teks, isError = false) {
      const el = document.getElementById("statusMsg");
      el.textContent = teks;
      el.style.color = isError ? "#f87171" : "#4ade80";
    }

    async function ambilBarang() {
      try {
        const res = await fetch(`${TABLE_ENDPOINT}?select=*&order=id.asc`, { headers });
        if (!res.ok) {
          const errText = await res.text();
          throw new Error(errText || `HTTP ${res.status}`);
        }
        daftarBarang = await res.json();
        renderTabel();
        setStatus("Terhubung ke database Supabase (tabel: barang).");
      } catch (err) {
        setStatus("Gagal memuat data dari Supabase. Cek RLS policy pada tabel 'barang'. Detail: " + err.message, true);
      }
    }

    function renderTabel() {
      const keyword = document.getElementById("cariInput").value.trim().toLowerCase();
      const filtered = daftarBarang.filter((item) =>
        String(item.nama || "").toLowerCase().includes(keyword) ||
        String(item.kategori || "").toLowerCase().includes(keyword)
      );

      const tbody = document.getElementById("tabelBarangBody");
      tbody.innerHTML = "";

      filtered.forEach((item, index) => {
        const tr = document.createElement("tr");
        tr.innerHTML = `
          <td>${index + 1}</td>
          <td><strong>${escapeHtml(item.nama)}</strong></td>
          <td>${escapeHtml(item.kategori || "-")}</td>
          <td>${escapeHtml(item.lokasi || "-")}</td>
          <td>${Number(item.jumlah) || 0}</td>
          <td>
            <button class="aksi-btn" onclick="ubahStok(${item.id}, 1)">+1</button>
            <button class="aksi-btn" onclick="ubahStok(${item.id}, -1)">-1</button>
            <button class="aksi-btn hapus" onclick="hapusBarang(${item.id})">Hapus</button>
          </td>
        `;
        tbody.appendChild(tr);
      });

      const totalItem = filtered.length;
      const totalStok = filtered.reduce((sum, item) => sum + (Number(item.jumlah) || 0), 0);
      document.getElementById("totalItem").textContent = totalItem;
      document.getElementById("totalStok").textContent = totalStok;
    }

    document.getElementById("formBarang").addEventListener("submit", async (e) => {
      e.preventDefault();
      const nama = document.getElementById("nama").value.trim();
      const kategori = document.getElementById("kategori").value;
      const jumlah = Math.max(0, parseInt(document.getElementById("jumlah").value, 10) || 0);
      const lokasi = document.getElementById("lokasi").value.trim();

      if (!nama) return;

      try {
        const res = await fetch(TABLE_ENDPOINT, {
          method: "POST",
          headers,
          body: JSON.stringify({ nama, kategori, jumlah, lokasi })
        });
        if (!res.ok) throw new Error(await res.text());
        document.getElementById("formBarang").reset();
        document.getElementById("jumlah").value = 1;
        await ambilBarang();
        setStatus("Barang berhasil disimpan ke Supabase.");
      } catch (err) {
        setStatus("Gagal menyimpan barang: " + err.message, true);
      }
    });

    async function ubahStok(id, delta) {
      const target = daftarBarang.find((x) => x.id === id);
      if (!target) return;
      const jumlahBaru = Math.max(0, (Number(target.jumlah) || 0) + delta);

      try {
        const res = await fetch(`${TABLE_ENDPOINT}?id=eq.${id}`, {
          method: "PATCH",
          headers,
          body: JSON.stringify({ jumlah: jumlahBaru })
        });
        if (!res.ok) throw new Error(await res.text());
        await ambilBarang();
      } catch (err) {
        setStatus("Gagal mengubah stok: " + err.message, true);
      }
    }

    async function hapusBarang(id) {
      try {
        const res = await fetch(`${TABLE_ENDPOINT}?id=eq.${id}`, {
          method: "DELETE",
          headers
        });
        if (!res.ok) throw new Error(await res.text());
        await ambilBarang();
      } catch (err) {
        setStatus("Gagal menghapus barang: " + err.message, true);
      }
    }

    function escapeHtml(text) {
      return String(text ?? "")
        .replace(/&/g, "&amp;")
        .replace(/</g, "&lt;")
        .replace(/>/g, "&gt;")
        .replace(/"/g, "&quot;");
    }

    document.getElementById("cariInput").addEventListener("input", renderTabel);

    ambilBarang();
  </script>
